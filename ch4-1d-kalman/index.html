from collections import namedtuple
gaussian = namedtuple('Gaussian', ['mean', 'var'])
gaussian.__repr__ = lambda s: 'N(mean={:.3f}, var={:.3f})'.format(s[0], s[1])
def predict(pos, movement):
    return gaussian(pos.mean + movement.mean, pos.var + movement.var)
x = gaussian(10., 0.2**2)    # x_{k-1}
fx = gaussian(15., 0.7**2)   # movement f_x
prior = predict(x, fx)

Update: Product of Gaussians08

update() Code

  • 두 Gaussian 의 곱으로 posterior 계산, 평균은 분산으로 가중한 합
  • 결과 , 측정 쪽으로 이동하고 분산 감소
  • 조건: prior , z 
def gaussian_multiply(g1, g2):
    mean = (g1.var * g2.mean + g2.var * g1.mean) / (g1.var + g2.var)
    variance = (g1.var * g2.var) / (g1.var + g2.var)
    return gaussian(mean, variance)
def update(prior, likelihood):
    posterior = gaussian_multiply(likelihood, prior)
    return posterior
prior = gaussian(25., 0.53)  # result of predict
z = gaussian(23., 0.16)      # measurement
posterior = update(prior, z)
print(posterior)

def product_var(var1, var2):
    return 1. / (1./var1 + 1./var2)
for v1, v2 in ((1, 100), (1, 1)):
    print('var1={}, var2={} -> var={:.3f}'.format(
          v1, v2, product_var(v1, v2)))

Summary: Kalman Form14

Kalman Form Code

  • residual y 와 Kalman gain K 로 다시 쓴 predict, update
  • Gaussian 곱으로 구한 결과와 같은 값
  • 조건: x , f_x , z 
def update(prior, measurement):
    x, P = prior        # mean and variance of prior
    z, R = measurement  # mean and variance of measurement
    y = z - x           # residual
    K = P / (P + R)     # Kalman gain
    x = x + K*y         # posterior
    P = (1 - K) * P     # posterior variance
    return gaussian(x, P)
def predict(posterior, movement):
    x, P = posterior    # mean and variance of posterior
    dx, Q = movement    # mean and variance of movement
    x = x + dx
    P = P + Q
    return gaussian(x, P)
prior = predict(gaussian(10., 0.2**2), gaussian(15., 0.7**2))
print(prior)
print(update(prior, gaussian(23., 0.16)))

def converged_variance(process_var, sensor_var, N=50):
    x = gaussian(0., 400.)
    process_model = gaussian(1., process_var)
    ps = []
    for _ in range(N):
        prior = predict(x, process_model)
        x = update(prior, gaussian(0., sensor_var))
        ps.append(x.var)
    return ps
for q, r in ((40, 2), (2, 40)):
    ps = converged_variance(q, r)
    plt.plot(ps, label='process_var={}, sensor_var={}'.format(q, r))
    print(q, r, '->', round(ps[-1], 3))
